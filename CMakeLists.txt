cmake_minimum_required(VERSION 3.20)
project(Unbound LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UNBOUND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
    add_subdirectory(src/platforms/windows)
elseif(APPLE)
    add_subdirectory(src/platforms/macos)
else()
    message(FATAL_ERROR "Unbound has no backend for ${CMAKE_SYSTEM_NAME}")
endif()
