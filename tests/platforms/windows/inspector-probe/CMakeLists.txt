cmake_minimum_required(VERSION 3.20)
project(InspectorProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Inspector probe requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Widgets)
add_library(InspectorProbe SHARED snapshot.cpp)
target_link_libraries(InspectorProbe PRIVATE Qt5::Widgets)
add_executable(inspector-probe WIN32 launcher.cpp)
set_property(TARGET inspector-probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(inspector-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNBOUND_VERSION=L"InspectorProbe")
target_link_libraries(inspector-probe PRIVATE version)
