cmake_minimum_required(VERSION 3.20)
project(UnboundDiagnostic LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Unbound diagnostics require Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Widgets)
get_filename_component(shared "${CMAKE_CURRENT_SOURCE_DIR}/../../../../src" ABSOLUTE)
include("${shared}/../infrastructure/windows/version.cmake")
string(APPEND UNBOUND_VERSION "-diagnostic")
set(UNBOUND_FILE_TYPE VFT_APP)
set(UNBOUND_FILENAME Unbound.exe)
configure_file("${shared}/platforms/windows/resources/version.rc.in" launcher.rc @ONLY)
set(UNBOUND_FILE_TYPE VFT_DLL)
set(UNBOUND_FILENAME Unbound.dll)
configure_file("${shared}/platforms/windows/resources/version.rc.in" helper.rc @ONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" "1 ICON \"${shared}/platforms/windows/resources/unbound.ico\"\n")
add_executable(Unbound WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc" "${CMAKE_CURRENT_BINARY_DIR}/icon.rc")
set_property(TARGET Unbound PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(Unbound PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNBOUND_VERSION=L"${UNBOUND_VERSION}")
target_link_libraries(Unbound PRIVATE version)
add_library(UnboundHelper SHARED controller.cpp "${CMAKE_CURRENT_BINARY_DIR}/helper.rc")
set_target_properties(UnboundHelper PROPERTIES OUTPUT_NAME Unbound)
set_property(TARGET UnboundHelper PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
include("${shared}/platforms/windows/windowing.cmake")
target_link_libraries(UnboundHelper PRIVATE UnboundWindowsWindowing)
add_executable(diagnostic-smoke main.cpp)
target_link_libraries(diagnostic-smoke PRIVATE Qt5::Widgets)
