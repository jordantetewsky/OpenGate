cmake_minimum_required(VERSION 3.20)
project(PaneFloatSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(UNBOUND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(WIN32)
    include("${UNBOUND_ROOT}/src/platforms/windows/windowing.cmake")
    add_executable(pane-float-smoke main.cpp
        "${UNBOUND_ROOT}/tests/platforms/windows/windowing/platformtests.cpp")
    target_include_directories(pane-float-smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(pane-float-smoke PRIVATE UnboundWindowsWindowing)
    add_executable(audition-activation-test auditionactivationtest.cpp)
    target_link_libraries(audition-activation-test PRIVATE UnboundWindowsWindowing)
else()
    message(FATAL_ERROR "Pane fixtures need a platform window backend and test adapter. Only Windows is implemented; see doc/platforms/macos.md.")
endif()
